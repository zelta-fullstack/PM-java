andi ingin membeli tumbler ikea seharga 280000, dan jika dia membuat member akan mendapatkan diskon 10%. di tambah jika dia membeli lebih dari sama dengan 250000, dia akan mendapatkan gratis ongkir sebesar 15000. Tapi jika belanjaan andi kurang dari 250000. dia harus membayar 15000.

start

let hargaBarang = 280000
let item = 2
let member = true
let total = hargaBarang * item
let diskon = total * 0.1

SWITCH input
	CASE FALSE:
	 msgInput = "input tidak valid"
	DEFAULT 
	 msgInput = "input valid" 

SWITCH member
    CASE TRUE:
        subtotal = total - diskon
    DEFAULT:
        subtotal = total

switch subtotal
    CASE true
        subtotal < 250000
        totalBayar = subtotal + 15000
    DEFAULT
	totalBayar = subtotal

display  totalBayar
END

jika ya lanjut jika tidak balik lagi ke awal, terus membuat total dari hargaBarang * item
terus membuat diskon dari total * 0.1 terus masuk ke pengecekan member jika true,
total - diskon dan dimasukan ke subtotal. jika false tampilkan total dan masukan ke subtotal.
setalh itu pengecekan subtotal jika kurang dari 250000 maka subtotal + 15000 dan dimasukan ke
totalBayar jika lebih dari 250000 tampilkan subtotal saja dan masukan ke totalBayar dan end.
